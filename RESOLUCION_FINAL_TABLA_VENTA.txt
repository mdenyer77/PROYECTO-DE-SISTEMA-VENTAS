        IF EXISTS (SELECT 1 FROM sys.objects WHERE type = 'P' 
                   AND name = 'sp_listar_venta')
            DROP PROCEDURE sp_listar_venta
        GO

        CREATE PROCEDURE sp_listar_venta
        AS
        BEGIN
            SELECT 
                ID_venta,
                Fecha_venta,
                ID_cliente,
                Total_general,
                Estado_venta
            FROM venta
            ORDER BY Fecha_venta DESC
        END
        GO
